     1→"use client";
     2→
     3→import { useEffect, useRef, useState, useCallback } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import {
     6→  ArrowLeft,
     7→  Send,
     8→  Loader2,

    10→  CheckCircle2,

    21→} from "lucide-react";
    22→import { useGameStore, type ChatMessage } from "@/lib/game/store";
    23→import type { NpcProfile } from "@/lib/game/npcs";
    24→import { callAgnes } from "@/lib/agnes/engine";
    25→import { QuickPhrases } from "./QuickPhrases";

    28→
    29→function genId() {
    30→  return Math.random().toString(36).slice(2) + Date.now().toString(36);
    31→}
    32→
    33→export function ChatWindow({ npc, onBack }: { npc: NpcProfile; onBack: () => void }) {
    34→  const conv = useGameStore((s) => s.conversations[npc.id]);
    35→  const appendMessage = useGameStore((s) => s.appendMessage);
    36→  const updateDefense = useGameStore((s) => s.updateDefense);
    37→  const setConversationStatus = useGameStore((s) => s.setConversationStatus);
    38→  const updateConversationMetrics = useGameStore((s) => s.updateConversationMetrics);
    39→  const resetConversation = useGameStore((s) => s.resetConversation);
    40→  const consumeTraffic = useGameStore((s) => s.consumeTraffic);
    41→  const dataTraffic = useGameStore((s) => s.dataTraffic);
    42→  const theme = useGameStore((s) => s.theme);
    43→  const showTimestamps = useGameStore((s) => s.showTimestamps);
    44→
    45→  const [input, setInput] = useState("");
    46→  const [thinking, setThinking] = useState(false);
    47→  const [successAnim, setSuccessAnim] = useState(false);
    48→  const [showResetConfirm, setShowResetConfirm] = useState(false);
    49→  const [showEnding, setShowEnding] = useState(false);
    50→  const [showQuickPhrases, setShowQuickPhrases] = useState(false);
    51→  const [showImageMaterials, setShowImageMaterials] = useState(false);
    52→  const [showNpcInfo, setShowNpcInfo] = useState(false);
    53→  const [hasReset, setHasReset] = useState(false);
    54→  const [failedMessages, setFailedMessages] = useState<Set<string>>(new Set());
    55→  const [aiVerified, setAiVerified] = useState(false); // 第一次成功後設為 true，之後不再驗證
    56→  const [keyboardHeight, setKeyboardHeight] = useState(0);
    57→  const scrollRef = useRef<HTMLDivElement>(null);
    58→  const inputRef = useRef<HTMLTextAreaElement>(null);
    59→  const userScrolledUpRef = useRef(false);
    60→  const lastScrollTsRef = useRef(0);
    61→
    62→  const scrollToBottom = useCallback((force = false) => {
    63→    const now = Date.now();
    64→    if (!force && now - lastScrollTsRef.current < 100) return;
    65→    lastScrollTsRef.current = now;
    66→    if (!scrollRef.current) return;
    67→    const el = scrollRef.current;
    68→    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight;
    69→    if (distanceFromBottom < 150) userScrolledUpRef.current = false;
    70→    if (!force && userScrolledUpRef.current) return;
    71→    el.scrollTop = el.scrollHeight;
    72→  }, []);
    73→
    74→  useEffect(() => {
    75→    const el = scrollRef.current;
    76→    if (!el) return;
    77→    const handleScroll = () => {
    78→      const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight;
    79→      userScrolledUpRef.current = distanceFromBottom > 150;
    80→    };
    81→    el.addEventListener("scroll", handleScroll, { passive: true });
    82→    return () => el.removeEventListener("scroll", handleScroll);
    83→  }, []);
    84→
    85→  useEffect(() => {
    86→    if (typeof window === "undefined") return;
    87→    const handleResize = () => {
    88→      if (window.visualViewport) {
    89→        const kbHeight = window.innerHeight - window.visualViewport.height;
    90→        setKeyboardHeight(kbHeight > 50 ? kbHeight : 0);
    91→      }
    92→      setTimeout(() => scrollToBottom(false), 50);
    93→      setTimeout(() => scrollToBottom(false), 200);
    94→    };
    95→    window.addEventListener("resize", handleResize);
    96→    window.addEventListener("orientationchange", handleResize);
    97→    if (window.visualViewport) {
    98→      window.visualViewport.addEventListener("resize", handleResize);
    99→      window.visualViewport.addEventListener("scroll", handleResize);
   100→    }
   101→    return () => {
   102→      window.removeEventListener("resize", handleResize);
   103→      window.removeEventListener("orientationchange", handleResize);
   104→      if (window.visualViewport) {
   105→        window.visualViewport.removeEventListener("resize", handleResize);
   106→        window.visualViewport.removeEventListener("scroll", handleResize);
   107→      }
   108→    };
   109→  }, [scrollToBottom]);
   110→
   111→  useEffect(() => {
   112→    requestAnimationFrame(() => scrollToBottom(false));
   113→  }, [conv?.messages.length, thinking, scrollToBottom]);
   114→
   115→  const handleInputFocus = () => {
   116→    setTimeout(() => scrollToBottom(true), 100);
   117→    setTimeout(() => scrollToBottom(true), 300);
   118→  };
   119→
   120→  // 對話結束時自動彈出結果面板
   121→  useEffect(() => {
   122→    if (conv?.status && conv.status !== "active" && conv.endingReason) {
   123→      // 對話剛結束 → 顯示結果面板
   124→      const lastMsg = conv.messages[conv.messages.length - 1];
   125→      if (lastMsg?.meta?.decision === "agree" || lastMsg?.meta?.decision === "block" || lastMsg?.meta?.decision === "cautious") {
   126→        setTimeout(() => setShowEnding(true), 1500);
   127→      }
   128→    }
   129→  }, [conv?.status, conv?.endingReason, conv?.messages]);
   130→
   131→  if (!conv) {
   132→    return (
   133→      <div className="h-full flex items-center justify-center text-[color:var(--im-bubble-system-text)] text-sm">
   134→        載入中...
   135→      </div>
   136→    );
   137→  }
   138→
   139→  const isLocked = conv.status !== "active";
   140→
   141→  const sendMessage = async (text: string, imageMaterial?: { type: string; label: string }, retryMsgId?: string) => {
   142→    const trimmed = (text || "").trim();
   143→    if (!trimmed && !imageMaterial) return;
   144→    if (thinking) return;
   145→    const latestConv = useGameStore.getState().conversations[npc.id];
   146→    if (!latestConv || latestConv.status !== "active") return;
   147→
   148→    // 流量檢查：每則訊息消耗 100MB
   149→    const TRAFFIC_PER_MSG = 100;
   150→    if (!consumeTraffic(TRAFFIC_PER_MSG)) {
   151→      // 流量不足，號碼被封
   152→      const blockMsg: ChatMessage = {
   153→        id: genId(),
   154→        role: "system",
   155→        content: "⚠ 流量耗盡！此號碼已被電信商封鎖，對話強制中斷。請至情報販子補給站購買免洗 SIM 卡。",
   156→        ts: Date.now(),
   157→      };
   158→      appendMessage(npc.id, blockMsg);
   159→      setConversationStatus(npc.id, "blocked", undefined, "流量耗盡，號碼被封鎖。");
   160→      return;
   161→    }
   162→
   163→    // 流量低於 500MB 時發送警告通知
   164→    const remainingTraffic = useGameStore.getState().dataTraffic;
   165→    if (remainingTraffic <= 500 && remainingTraffic > 0) {
   166→      const warnMsg: ChatMessage = {
   167→        id: genId(),
   168→        role: "system",
   169→        content: `⚠ 流量警告：剩餘 ${(remainingTraffic / 1000).toFixed(1)} GB，號碼即將被封鎖！`,
   170→        ts: Date.now(),
   171→      };
   172→      appendMessage(npc.id, warnMsg);
   173→    }
   174→
   175→    setInput("");
   176→    setShowQuickPhrases(false);
   177→    setShowImageMaterials(false);
   178→    setThinking(true);
   179→
   180→    const finalText = imageMaterial ? `[圖片：${imageMaterial.label}] ${trimmed}` : trimmed;
   181→    const playerMsg: ChatMessage = {
   182→      id: genId(),
   183→      role: "player",
   184→      content: finalText,
   185→      ts: Date.now(),
   186→      meta: imageMaterial ? { imageType: imageMaterial.type } : undefined,
   187→    };
   188→    appendMessage(npc.id, playerMsg);
   189→
   190→    // 偵測訊號
   191→    const moneyCues = ["轉帳", "匯款", "帳戶", "ATM", "OTP", "驗證碼", "轉過去", "匯過去", "繳交", "支付", "付款", "繳費", "存入"];
   192→    const urgentCues = ["急", "快", "現在", "馬上", "立刻", "限時", "今天內"];
   193→    const isUrgent = urgentCues.some((k) => finalText.includes(k));
   194→    const isMoney = moneyCues.some((k) => finalText.includes(k));
   195→
   196→    // 更新連續計數 + 輪數
   197→    updateConversationMetrics(npc.id, isUrgent, isMoney);
   198→
   199→    const currentMessages = latestConv.messages;
   200→    const historyForAI = currentMessages
   201→      .filter((m) => m.role === "player" || m.role === "npc")
   202→      .map((m) => ({
   203→        role: m.role === "player" ? ("player" as const) : ("npc" as const),
   204→        content: m.content,
   205→      }));
   206→
   207→    // 重新讀取更新後的 conv（含最新 metrics）
   208→    const updatedConv = useGameStore.getState().conversations[npc.id];
   209→    if (!updatedConv) return;
   210→
   211→    try {
   212→      const data = await callAgnes({
   213→        sessionId: `${npc.id}-${updatedConv.startedAt}`,
   214→        npc,
   215→        playerMessage: finalText,
   216→        currentDefense: updatedConv.defense,
   217→        history: historyForAI,
   218→        consecutiveUrgent: updatedConv.consecutiveUrgent,
   219→        consecutiveMoney: updatedConv.consecutiveMoney,
   220→        turns: updatedConv.turns,
   221→      });
   222→
   223→      // AI 連線成功 → 標記為已驗證，之後不再驗證
   224→      setAiVerified(true);
   225→
   226→      if (data.defenseDelta) {
   227→        updateDefense(npc.id, data.defenseDelta);
   228→      }
   229→
   230→      // 不再有人工延遲 — AI 回應後立即顯示
   231→
   232→      const npcMsg: ChatMessage = {
   233→        id: genId(),
   234→        role: "npc",
   235→        content: data.reply,
   236→        ts: Date.now(),
   237→        meta: { decision: data.decision },
   238→      };
   239→      appendMessage(npc.id, npcMsg);
   240→
   241→      // 處理多結局
   242→      if (data.decision === "agree" && data.payoutAmount) {
   243→        const sysMsg: ChatMessage = {
   244→          id: genId(),
   245→          role: "system",
   246→          content: `✓ ${npc.displayName} 已同意轉帳 $${data.payoutAmount.toLocaleString()}。款項已到帳。`,
   247→          ts: Date.now(),
   248→          meta: { decision: "agree", amount: data.payoutAmount },
   249→        };
   250→        appendMessage(npc.id, sysMsg);
   251→        setConversationStatus(npc.id, "succeeded", data.payoutAmount, data.endingReason);
   252→        setSuccessAnim(true);
   253→        setTimeout(() => setSuccessAnim(false), 2500);
   254→      } else if (data.decision === "block") {
   255→        const sysMsg: ChatMessage = {
   256→          id: genId(),
   257→          role: "system",
   258→          content: `✗ ${npc.displayName} 已將你封鎖。對話終止。`,
   259→          ts: Date.now(),
   260→          meta: { decision: "block" },
   261→        };
   262→        appendMessage(npc.id, sysMsg);
   263→        setConversationStatus(npc.id, "blocked", undefined, data.endingReason);
   264→      } else if (data.decision === "cautious") {
   265→        const sysMsg: ChatMessage = {
   266→          id: genId(),
   267→          role: "system",
   268→          content: `⚠ ${npc.displayName} 不願再繼續討論這件事。對話結束。`,
   269→          ts: Date.now(),
   270→          meta: { decision: "cautious" },
   271→        };
   272→        appendMessage(npc.id, sysMsg);
   273→        setConversationStatus(npc.id, "cautious", undefined, data.endingReason);
   274→      }
   275→    } catch (e) {
   276→      console.error("[ChatWindow] callAgnes failed:", e);
   277→      // 只有在尚未驗證通過時才顯示「發送失敗」
   278→      // 已驗證通過後的臨時失敗不顯示（避免打擾玩家對話流暢度）
   279→      if (!aiVerified) {
   280→        setFailedMessages(prev => new Set(prev).add(playerMsg.id));
   281→      } else {
   282→        // 已驗證但臨時失敗 → 顯示系統提示但不阻斷
   283→        const errMsg: ChatMessage = {
   284→          id: genId(),
   285→          role: "system",
   286→          content: "⚠ 訊息發送延遲，請稍候再試。",
   287→          ts: Date.now(),
   288→        };
   289→        appendMessage(npc.id, errMsg);
   290→      }
   291→    } finally {
   292→      setThinking(false);
   293→      setTimeout(() => inputRef.current?.focus(), 100);
   294→    }
   295→  };
   296→
   297→  const handleReset = () => {
   298→    if (hasReset) return;
   299→    resetConversation(npc.id);
   300→    setHasReset(true);
   301→    setAiVerified(false); // 重置後重新需要驗證
   302→    setFailedMessages(new Set()); // 清除失敗標記
   303→    setShowResetConfirm(false);
   304→    setShowEnding(false);
   305→    setThinking(false);
   306→    setSuccessAnim(false);
   307→    setTimeout(() => inputRef.current?.focus(), 100);
   308→  };
   309→
   310→  const handleExport = () => {
   311→    const lines: string[] = [];
   312→    lines.push(`# SimOS 對話記錄 - ${npc.displayName}`);
   313→    lines.push(`# 匯出時間：${new Date().toLocaleString("zh-TW")}`);
   314→    lines.push(`# NPC: ${npc.displayName} (${npc.id})`);
   315→    lines.push(`# TeleChat ID: ${npc.telechatId}`);
   316→    lines.push(`# 對話狀態：${conv.status}`);
   317→    if (conv.payout) lines.push(`# 詐騙金額：$${conv.payout.toLocaleString()}`);
   318→    if (conv.endingReason) lines.push(`# 結局：${conv.endingReason}`);
   319→    lines.push(`# 對話輪數：${conv.turns}`);
   320→    lines.push("");
   321→    lines.push("---");
   322→    lines.push("");
   323→    for (const msg of conv.messages) {
   324→      const time = new Date(msg.ts).toLocaleString("zh-TW");
   325→      let speaker: string;
   326→      if (msg.role === "player") speaker = "🧑 玩家";
   327→      else if (msg.role === "npc") speaker = `${npc.avatar} ${npc.displayName}`;
   328→      else speaker = "ℹ️ 系統";
   329→      lines.push(`[${time}] ${speaker}:`);
   330→      lines.push(msg.content);
   331→      lines.push("");
   332→    }
   333→    const blob = new Blob([lines.join("\n")], { type: "text/plain;charset=utf-8" });
   334→    const url = URL.createObjectURL(blob);
   335→    const a = document.createElement("a");
   336→    a.href = url;
   337→    a.download = `SimOS_${npc.id}_${Date.now()}.txt`;
   338→    document.body.appendChild(a);
   339→    a.click();
   340→    document.body.removeChild(a);
   341→    URL.revokeObjectURL(url);
   342→  };
   343→
   344→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   345→    if (e.key === "Enter" && !e.shiftKey) {
   346→      e.preventDefault();
   347→      sendMessage(input);
   348→    }
   349→  };
   350→
   351→  const canSend = input.trim().length > 0 && !thinking && !isLocked;
   352→
   353→  return (
   354→    <div
   355→      className="h-full min-h-0 flex flex-col relative overflow-hidden"
   356→      style={{ background: "var(--im-bg)" }}
   357→    >
   358→      {/* iOS iMessage 頂部導航欄 */}
   359→      <div
   360→        className="flex items-center px-2 py-2 border-b backdrop-blur-xl z-20 shrink-0"
   361→        style={{
   362→          background: "var(--im-header-bg)",
   363→          borderColor: "var(--im-header-border)",
   364→          color: "var(--im-header-text)",
   365→        }}
   366→      >
   367→        <button
   368→          onClick={onBack}
   369→          className="flex items-center gap-1 text-[17px] font-normal px-1 active:opacity-50 transition"
   370→          style={{ color: "var(--im-link-text)" }}
   371→          aria-label="返回"
   372→        >
   373→          <ChevronUp className="w-5 h-5 rotate-[-90deg]" />
   374→          <span className="max-w-[60px] truncate">返回</span>
   375→        </button>
   376→
   377→        <div className="flex-1 flex flex-col items-center justify-center">
   378→          <button
   379→            onClick={() => setShowNpcInfo(true)}
   380→            className="flex items-center gap-1.5 active:scale-95 transition"
   381→          >
   382→            <div
   383→              className="w-7 h-7 rounded-full flex items-center justify-center text-base"
   384→              style={{ background: "var(--im-bubble-npc-bg)" }}
   385→            >
   386→              {npc.avatar}
   387→            </div>
   388→            <div className="flex flex-col items-start">
   389→              <span className="text-[16px] font-semibold truncate max-w-[140px]" style={{ color: "var(--im-header-text)" }}>
   390→                {npc.displayName}
   391→              </span>
   392→              {failedMessages.size > 0 && (
   393→                <span className="text-[10px] font-medium" style={{ color: "#ff3b30" }}>
   394→                  連接網絡失敗
   395→                </span>
   396→              )}
   397→            </div>
   398→          </button>
   399→        </div>
   400→
   401→        <div className="flex items-center gap-1 w-[80px] justify-end">
   402→          <button
   403→            onClick={handleExport}
   404→            className="p-1.5 rounded-full active:opacity-50 transition"
   405→            style={{ color: "var(--im-link-text)" }}
   406→            aria-label="匯出"
   407→            title="匯出對話"
   408→          >
   409→            <Download className="w-5 h-5" />
   410→          </button>
   411→          {!hasReset && (
   412→            <button
   413→              onClick={() => setShowResetConfirm(true)}
   414→              className="p-1.5 rounded-full active:opacity-50 transition"
   415→              style={{ color: "var(--im-link-text)" }}
   416→              aria-label="重新開始"
   417→              title="重新開始對話（僅限一次）"
   418→            >
   419→              <RotateCcw className="w-5 h-5" />
   420→            </button>
   421→          )}
   422→        </div>
   423→      </div>
   424→
   425→      {/* 成功動畫 */}
   426→      <AnimatePresence>
   427→        {successAnim && (
   428→          <motion.div
   429→            initial={{ opacity: 0, scale: 0.5 }}
   430→            animate={{ opacity: 1, scale: 1 }}
   431→            exit={{ opacity: 0, scale: 0.5 }}
   432→            className="absolute inset-0 z-50 flex items-center justify-center bg-black/70 backdrop-blur-sm"
   433→          >
   434→            <div className="flex flex-col items-center gap-3">
   435→              <motion.div
   436→                initial={{ rotate: -180, scale: 0 }}
   437→                animate={{ rotate: 0, scale: 1 }}
   438→                transition={{ type: "spring", stiffness: 200 }}
   439→                className="w-24 h-24 rounded-full bg-gradient-to-br from-amber-400 to-yellow-600 flex items-center justify-center shadow-2xl"
   440→              >
   441→                <Trophy className="w-12 h-12 text-white" />
   442→              </motion.div>
   443→              <motion.div
   444→                initial={{ opacity: 0, y: 10 }}
   445→                animate={{ opacity: 1, y: 0 }}
   446→                transition={{ delay: 0.2 }}
   447→                className="text-center"
   448→              >
   449→                <p className="text-white text-xl font-bold">詐騙成功！</p>
   450→                <p className="text-amber-400 text-2xl font-bold mt-1">
   451→                  +${conv.payout?.toLocaleString()}
   452→                </p>
   453→              </motion.div>
   454→            </div>
   455→          </motion.div>
   456→        )}
   457→      </AnimatePresence>
   458→
   459→      {/* 訊息列表 - iMessage 風格 */}
   460→      <div ref={scrollRef} className="flex-1 min-h-0 overflow-y-auto px-3 py-3 space-y-0.5">
   461→        {conv.messages.map((msg, idx) => (
   462→          <MessageBubble
   463→            key={msg.id}
   464→            msg={msg}
   465→            npcAvatar={npc.avatar}
   466→            showTimestamp={showTimestamps}
   467→            prevMsg={idx > 0 ? conv.messages[idx - 1] : undefined}
   468→            onAvatarClick={() => setShowNpcInfo(true)}
   469→            isFailed={failedMessages.has(msg.id)}
   470→            onRetry={() => {
   471→              // 重新發送：移除失敗標記，直接重試
   472→              setFailedMessages(prev => {
   473→                const next = new Set(prev);
   474→                next.delete(msg.id);
   475→                return next;
   476→              });
   477→              sendMessage(msg.content, undefined, msg.id);
   478→            }}
   479→          />
   480→        ))}
   481→        {thinking && (
   482→          <div className="flex items-end gap-1.5 px-2 my-1">
   483→            <div className="w-7 h-7 rounded-full flex items-center justify-center text-base shrink-0" style={{ background: "var(--im-bubble-npc-bg)" }}>
   484→              {npc.avatar}
   485→            </div>
   486→            <div className="px-3 py-2.5" style={{ background: "var(--im-bubble-npc-bg)", borderRadius: "18px 18px 18px 4px" }}>
   487→              <div className="flex gap-1">
   488→                {[0, 1, 2].map((i) => (
   489→                  <motion.div
   490→                    key={i}
   491→                    animate={{ opacity: [0.3, 1, 0.3] }}
   492→                    transition={{ duration: 1, repeat: Infinity, delay: i * 0.15 }}
   493→                    className="w-1.5 h-1.5 rounded-full"

   697→function MessageBubble({
   698→  msg,
   699→  npcAvatar,
   700→  showTimestamp,
   701→  prevMsg,
   702→  onAvatarClick,
   703→  isFailed,
   704→  onRetry,
   705→}: {
   706→  msg: ChatMessage;
   707→  npcAvatar: string;
   708→  showTimestamp: boolean;
   709→  prevMsg?: ChatMessage;
   710→  onAvatarClick?: () => void;
   711→  isFailed?: boolean;
   712→  onRetry?: () => void;
   713→}) {
   714→  // 系統訊息：置中，灰色圓角
   715→  if (msg.role === "system") {
   716→    const isAgree = msg.meta?.decision === "agree";
   717→    const isBlock = msg.meta?.decision === "block";
   718→    const isCautious = msg.meta?.decision === "cautious";
   719→    return (
   720→      <div className="flex justify-center my-2">
   721→        <div
   722→          className="text-[11px] px-3 py-1.5 rounded-full flex items-center gap-1.5 max-w-[85%] text-center"
   723→          style={{
   724→            background: isAgree ? "rgba(48, 209, 88, 0.15)" : isBlock ? "rgba(255, 69, 58, 0.15)" : isCautious ? "rgba(255, 159, 10, 0.15)" : "var(--im-bubble-system-bg)",
   725→            color: isAgree ? "#30d158" : isBlock ? "#ff453a" : isCautious ? "#ff9f0a" : "var(--im-bubble-system-text)",
   726→          }}
   727→        >
   728→          {isAgree && <CheckCircle2 className="w-3 h-3 shrink-0" />}
   729→          {isBlock && <UserX className="w-3 h-3 shrink-0" />}
   730→          {isCautious && <AlertCircle className="w-3 h-3 shrink-0" />}
   731→          {msg.content}
   732→        </div>
   733→      </div>
   734→    );
   735→  }
   736→
   737→  const isPlayer = msg.role === "player";
   738→  // 判斷是否為同一群組（連續同 role 5 分鐘內）
   739→  const isGroupStart = !prevMsg || prevMsg.role !== msg.role || (msg.ts - prevMsg.ts) > 5 * 60 * 1000;
   740→  // 顯示頭像 only 當 group end（最後一條同 role）
   741→  const showAvatar = !isPlayer;
   742→
   743→  // 時間戳顯示（每 group start 顯示）
   744→  const showTime = showTimestamp && isGroupStart;
   745→
   746→  return (
   747→    <motion.div
   748→      initial={{ opacity: 0, y: 4 }}
   749→      animate={{ opacity: 1, y: 0 }}
   750→      transition={{ duration: 0.15 }}
   751→      className={`flex items-end gap-1.5 ${isPlayer ? "flex-row-reverse" : ""} ${isGroupStart ? "mt-2" : "mt-0.5"}`}
   752→    >
   753→      {/* NPC 頭像 - 可點擊查看情報 */}
   754→      {showAvatar ? (
   755→        <button
   756→          onClick={() => onAvatarClick?.()}
   757→          className="w-7 h-7 rounded-full flex items-center justify-center text-base shrink-0 active:scale-90 transition"
   758→          style={{ background: "var(--im-bubble-npc-bg)" }}
   759→          aria-label="查看情報"
   760→        >
   761→          {npcAvatar}
   762→        </button>
   763→      ) : (
   764→        !isPlayer && <div className="w-7 shrink-0" />
   765→      )}
   766→
   767→      <div className={`flex flex-col ${isPlayer ? "items-end" : "items-start"} max-w-[75%]`}>
   768→        {showTime && (
   769→          <div className="text-[10px] mb-1 px-2" style={{ color: "var(--im-timestamp-text)" }}>
   770→            {new Date(msg.ts).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}
   771→          </div>
   772→        )}
   773→        <div
   774→          className="px-3.5 py-2 text-[15px] leading-[20px] whitespace-pre-wrap break-words"
   775→          style={{
   776→            background: isPlayer ? "var(--im-bubble-player-bg)" : "var(--im-bubble-npc-bg)",
   777→            color: isPlayer ? "var(--im-bubble-player-text)" : "var(--im-bubble-npc-text)",
   778→            borderRadius: isPlayer
   779→              ? isGroupStart ? "18px 18px 4px 18px" : "18px 4px 4px 18px"
   780→              : isGroupStart ? "18px 18px 18px 4px" : "4px 18px 18px 4px",
   781→          }}
   782→        >
   783→          {msg.meta?.imageType && (
   784→            <div className="mb-1.5 px-2 py-1 rounded bg-black/20 text-[10px] inline-block">
   785→              📎 圖片素材
   786→            </div>
   787→          )}
   788→          {msg.content}
   789→        </div>
   790→        {/* 發送失敗指示器 */}
   791→        {isFailed && isPlayer && (
   792→          <div className="flex items-center gap-1 mt-0.5">
   793→            <button
   794→              onClick={onRetry}
   795→              className="flex items-center gap-1 text-[11px] font-medium active:scale-95 transition"
   796→              style={{ color: "#ff3b30" }}
   797→            >
   798→              <AlertCircle className="w-3.5 h-3.5" />
   799→              發送失敗 · 點擊重試
   800→            </button>
   801→          </div>
   802→        )}
   803→      </div>
   804→    </motion.div>
   805→  );
   806→}
